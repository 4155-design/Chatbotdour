const http = require("http"); const PORT = process.env.PORT || 3000; const MODEL = "gpt-6-astra"; // Deinen Modellnamen beibehalten const MODES = { normal: "Du bist DOUR AI, ein intelligenter KI-Assistent. Antworte auf Deutsch, klar, freundlich und hilfreich.", code: "Du bist DOUR AI Coding Expert. Schreibe sauberen, vollständigen, gut strukturierten Code. Erkläre Fehler und liefere Lösungen.", creative: "Du bist DOUR AI Creative. Entwickle originelle Ideen, Konzepte, Texte und Projekte.", learn: "Du bist DOUR AI Tutor. Erkläre schwierige Themen einfach und Schritt für Schritt." }; const server = http.createServer(async (req, res) => { res.setHeader("Content-Type", "application/json; charset=utf-8"); res.setHeader("Access-Control-Allow-Origin", "*"); res.setHeader("Access-Control-Allow-Headers", "Content-Type"); res.setHeader("Access-Control-Allow-Methods", "POST, OPTIONS"); if (req.method === "OPTIONS") { res.writeHead(204); return res.end(); } if (req.method !== "POST" || req.url !== "/api/chat") { res.writeHead(404); return res.end(JSON.stringify({error:"Nicht gefunden"})); } try { if (!process.env.OPENAI_API_KEY) { throw new Error("KI-Schlüssel fehlt auf dem Server."); } let raw = ""; for await (const chunk of req) { raw += chunk; if (raw.length > 50000) throw new Error("Nachricht zu lang."); } const body = JSON.parse(raw); const messages = body.messages; if (!Array.isArray(messages)) { throw new Error("Ungültige Nachrichten."); } const mode = MODES[body.mode] ? body.mode : "normal"; const input = messages.slice(-20).map(m => ({ role: m.role, content: String(m.content).slice(0, 4000) })).filter(m => ["user", "assistant"].includes(m.role) ); const answer = await fetch( "https://api.openai.com/v1/responses", { method: "POST", headers: { "Authorization": `Bearer ${process.env.OPENAI_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: MODEL, instructions: MODES[mode], input }) } ); const data = await answer.json(); if (!answer.ok) { throw new Error("KI-API-Fehler: " + (data.error?.message || answer.status)); } const result = (data.output || []) .flatMap(x => x.content || []) .filter(x => x.type === "output_text") .map(x => x.text) .join("\n"); res.writeHead(200); res.end(JSON.stringify({ answer: result || "Keine Antwort erhalten." })); } catch (error) { res.writeHead(500); res.end(JSON.stringify({error:error.message})); } }); server.listen(PORT, "0.0.0.0", () => { console.log("DOUR AI Server gestartet auf Port " + PORT); });
